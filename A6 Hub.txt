-- Algorithm to encode Player's Input to Hex for matching
local function EncodeInput(str)
    local encoded = ""
    for i = 1, #str do
        encoded = encoded .. string.format("%02X", string.byte(str, i) + 15)
    end
    return encoded
end

-- [3] PREDICTOR & HOP ENGINE LOGIC
local function HopServer()
    local Api = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/0?sortOrder=Asc&limit=100"
    local Success, Result = pcall(function() return HttpService:JSONDecode(game:HttpGet(Api)) end)
    if Success and Result and Result.data then
        for _, server in pairs(Result.data) do
            if server.playing < server.maxPlayers and server.id ~= game.JobId then
                TeleportService:TeleportToPlaceInstance(game.PlaceId, server.id, LocalPlayer)
                break
            end
        end
    end
end

local function GetFullMoonPrediction()
    local moonTexture = Lighting.Sky.MoonTextureId
    local isFullMoon = moonTexture == "http://www.roblox.com/asset/?id=9709149431" or moonTexture == "rbxassetid://9709149431"
    if isFullMoon then return "[STATUS] FULL MOON IS ACTIVE!" end
    local timeInGame = Lighting.ClockTime
    if timeInGame < 18 and timeInGame >= 6 then
        return "[INFO] Next Full Moon in approx: ~" .. math.floor((18 - timeInGame) * 60 / 18) .. " mins"
    else
        return "[INFO] Waiting for the next night cycle..."
    end
end

local function GetFruitSpawnPrediction()
    local uptime = workspace.DistributedGameTime
    local spawnInterval = 3600
    local nextSpawnSeconds = spawnInterval - (uptime % spawnInterval)
    return string.format("[TIMER] Fruit spawns in: %02d mins %02d secs", math.floor(nextSpawnSeconds / 60), math.floor(nextSpawnSeconds % 60))
end

local function CheckHighTierFruitInServer()
    for _, v in pairs(workspace:GetChildren()) do
        if v:IsA("Tool") or string.find(v.Name, "Fruit") then
            for _, fruitName in pairs(HighTierFruits) do
                if string.find(v.Name, fruitName) then return true, v.Name end
            end
        end
    end
    return false, nil
end
